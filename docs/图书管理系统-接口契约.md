```java
package com.library.dao;

import java.util.List;

public interface IBaseDao<T> {
    /** 查询全部记录 */
    List<T> getList();

    /** 按主键查询，不存在返回 null */
    T getById(int id);

    /** 批量新增（list 中 id=0 的记录） */

    void deleteList(List<T> list);

    /** 按主键更新整条记录（对象所有字段覆盖写） */

}
```

```java
import java.util.List;
```

```java
public class BaseTableModel<T> extends AbstractTableModel {
    /**
     * @param list    数据源
     * @param columns 表头，如 {"编号","书名","分类","作者","库存"}
     * @param props   与 columns 一一对应的属性路径，如 {"id","name","category.name","author","stock"}
     *                支持用 "." 取嵌套属性（如 book.category.name）
     */
```

## 七、跨模块调用示例（联调口径）

**借书流程（D 的 BorrowPane）**

```java
// 1. 选读者（下拉，来自 B 的 ReaderDao）
Reader reader = readerDao.getByCardNo(cardNo);
// 2. 选图书（检索，来自 C 的 BookDao）
List<Book> books = bookDao.getByKeyword(keyword);
Book book = books.get(选中行);
// 3. 扣库存（只调接口，SQL 在 C 的实现里）
if (!bookDao.updateStock(book.getId(), -1)) {
    // 提示"库存不足"，流程终止
}
// 4. 生成借阅记录（D 自己的 DAO）
Borrow b = new Borrow();
b.setBorrowNo("20260914" + 序号);   // yyyyMMdd + 4 位序号
b.setReader(reader);
b.setBook(book);
b.setBorrowDate(new Date());
b.setDueDate(new Date(System.currentTimeMillis() + 30L * 24 * 3600 * 1000)); // +30 天
b.setStatus("借出");
borrowDao.addBorrow(b);
```

**还书流程（D 的 ReturnPane）**

```java
Borrow b = borrowDao.getById(borrowId);          // 或从 getBorrowingList() 选中行取
long overdueDays = (今天 - b.getDueDate().getTime()) / 86400000L;
double fine = overdueDays > 0 ? overdueDays * 0.5 : 0;   // 每逾期 1 天 0.5 元
borrowDao.returnBook(b.getId(), fine);            // D 自己的 DAO
bookDao.updateStock(b.getBook().getId(), +1);     // 恢复库存，调 C 的接口
```

**逾期检查（E 的 OverdueTimer）**

```java
List<Borrow> list = borrowDao.getBorrowingList();  // 只读调用 D 的接口
for (Borrow b : list) {
    if (b.getReturnDate() == null && b.getDueDate().before(今天)) {
        b.setStatus("逾期");
        borrowDao.update(b);                        // 更新状态
    }
}
```

**借阅排行（E 的 StatisticPane，集合知识点）**

```java
List<Borrow> all = borrowDao.getList();            // 只读调用

for (Borrow b : all) count.merge(b.getBook().getName(), 1, Integer::sum);
```
